
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

local SCAN_INTERVAL = 5
local TOUCH_MULTIPLIER = 6

local function isOtherPlayerContainer(container)
    if not container then
        return false
    end

    local owner = Players:GetPlayerFromCharacter(container)

    if owner then
        return owner ~= LocalPlayer
    end

    if container:IsA("Backpack") then
        local player = container.Parent
        return player:IsA("Player") and player ~= LocalPlayer
    end

    return false
end

local function scanTool(tool)
    if not tool:IsA("Tool") then
        return
    end

    if not isOtherPlayerContainer(tool.Parent) then
        return
    end

    for _, obj in ipairs(tool:GetDescendants()) do
        if obj:IsA("TouchTransmitter") then
            obj:Destroy()
        end
    end
end

local function setupLocalTool(tool)
    if not tool:IsA("Tool") then
        return
    end

    local function setupPart(obj)
        if not obj:IsA("BasePart") then
            return
        end

        if obj:GetAttribute("TripleTouchConnected") then
            return
        end

        obj:SetAttribute("TripleTouchConnected", true)

        for i = 1, TOUCH_MULTIPLIER do
            obj.Touched:Connect(function(hit)
               
            end)
        end
    end

    for _, obj in ipairs(tool:GetDescendants()) do
        setupPart(obj)
    end

    tool.DescendantAdded:Connect(setupPart)
end

local function scanPlayer(player)
    if player == LocalPlayer then
        return
    end

    local backpack = player:FindFirstChildOfClass("Backpack")
    if backpack then
        for _, obj in ipairs(backpack:GetChildren()) do
            if obj:IsA("Tool") then
                scanTool(obj)
            end
        end
    end

    local character = player.Character
    if character then
        for _, obj in ipairs(character:GetChildren()) do
            if obj:IsA("Tool") then
                scanTool(obj)
            end
        end
    end
end

local function scanAllPlayers()
    for _, player in ipairs(Players:GetPlayers()) do
        scanPlayer(player)
    end
end

local function watchContainer(container, isLocal)
    container.ChildAdded:Connect(function(child)
        if child:IsA("Tool") then
            task.defer(function()
                if isLocal then
                    setupLocalTool(child)
                else
                    scanTool(child)
                end
            end)
        end
    end)

    if not isLocal then
        container.DescendantAdded:Connect(function(obj)
            if obj:IsA("TouchTransmitter") then
                local tool = obj:FindFirstAncestorOfClass("Tool")
                if tool then
                    task.defer(scanTool, tool)
                end
            end
        end)
    end
end

local function watchPlayer(player)
    local isLocal = player == LocalPlayer

    local function onCharacter(character)
        watchContainer(character, isLocal)

        if isLocal then
            for _, obj in ipairs(character:GetChildren()) do
                if obj:IsA("Tool") then
                    setupLocalTool(obj)
                end
            end
        else
            task.defer(scanPlayer, player)
        end
    end

    player.CharacterAdded:Connect(onCharacter)

    local backpack = player:FindFirstChildOfClass("Backpack")
    if backpack then
        watchContainer(backpack, isLocal)

        if isLocal then
            for _, obj in ipairs(backpack:GetChildren()) do
                if obj:IsA("Tool") then
                    setupLocalTool(obj)
                end
            end
        end
    end

    player.ChildAdded:Connect(function(child)
        if child:IsA("Backpack") then
            watchContainer(child, isLocal)

            if isLocal then
                for _, obj in ipairs(child:GetChildren()) do
                    if obj:IsA("Tool") then
                        setupLocalTool(obj)
                    end
                end
            else
                task.defer(scanPlayer, player)
            end
        end
    end)

    if player.Character then
        onCharacter(player.Character)
    end
end

for _, player in ipairs(Players:GetPlayers()) do
    watchPlayer(player)
end

Players.PlayerAdded:Connect(watchPlayer)

while true do
    scanAllPlayers()
    task.wait(SCAN_INTERVAL)
end
