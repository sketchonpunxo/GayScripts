local function getTorsoPart(character)
    if not character then return nil end
    local candidates = {"HumanoidRootPart", "UpperTorso", "Torso"}
    for _, name in ipairs(candidates) do
        local p = character:FindFirstChild(name)
        if p and p:IsA("BasePart") then
            return p
        end
    end
    return nil
end

local function getArmParts(character)
    if not character then return {} end
    local arms = {}
    local r15Names = {"LeftUpperArm", "RightUpperArm", "LeftLowerArm", "RightLowerArm"}
    for _, name in ipairs(r15Names) do
        local p = character:FindFirstChild(name)
        if p and p:IsA("BasePart") then table.insert(arms, p) end
    end
    if #arms == 0 then
        local pL = character:FindFirstChild("Left Arm") or character:FindFirstChild("LeftArm")
        local pR = character:FindFirstChild("Right Arm") or character:FindFirstChild("RightArm")
        if pL and pL:IsA("BasePart") then table.insert(arms, pL) end
        if pR and pR:IsA("BasePart") then table.insert(arms, pR) end
    end
    return arms
end

local function classifyArms(armParts)
    local left, right
    for _, p in ipairs(armParts) do
        local name = p.Name:lower()
        if name:find("left") then left = p
        elseif name:find("right") then right = p
        end
    end
    if not left or not right then
        table.sort(armParts, function(a,b) return a.Position.X < b.Position.X end)
        if #armParts >= 1 and not left then left = armParts[1] end
        if #armParts >= 2 and not right then right = armParts[2] end
    end
    return left, right
end

local function computeOrbitParams(torsoPart, armParts)
    if not torsoPart then
        return MIN_ORBIT_RADIUS, 0
    end

    local localRef = nil
    if localPlayer and localPlayer.Character then
        localRef = getTorsoPart(localPlayer.Character)
    end

    local leftArm, rightArm = classifyArms(armParts)

    if not leftArm and not rightArm then
        return MIN_ORBIT_RADIUS, 0
    end

    local chooseArm = nil
    local chosenName = "unknown"
    if localRef and leftArm and rightArm then
        local dL = (leftArm.Position - localRef.Position).Magnitude
        local dR = (rightArm.Position - localRef.Position).Magnitude
        if dL <= dR then
            chooseArm = leftArm
            chosenName = "left"
        else
            chooseArm = rightArm
            chosenName = "right"
        end
    else
        if leftArm then
            chooseArm = leftArm
            chosenName = "left"
        else
            chooseArm = rightArm
            chosenName = "right"
        end
    end

    local torsoPos = torsoPart.Position
    local offset = chooseArm.Position - torsoPos
    local xz = math.sqrt(offset.X*offset.X + offset.Z*offset.Z)

    if chosenName == "right" and localRef then
        local distToLocal = (chooseArm.Position - localRef.Position).Magnitude
        if distToLocal > ARM_ONLY_DISTANCE then
            local radius = math.max(MIN_ORBIT_RADIUS, ARM_MARGIN)
            local yOffset = offset.Y
            return radius, yOffset
        end
    end

    local radius = math.max(MIN_ORBIT_RADIUS, xz + ARM_MARGIN)
    local yOffset = offset.Y

    return radius, yOffset
end

local function createDetachedClone(handle)
    if not handle or not handle:IsA("BasePart") then return nil end
    local clone = handle:Clone()
    clone.Parent = workspace
    clone.CanCollide = false
    pcall(function() clone.Massless = true end)
    clone.Anchored = true
    clone.CFrame = handle.CFrame
    return clone
end

local function startTrackingHandle(handle)
    if trackedHandles[handle] then return end
    if not handle or not handle.Parent then return end

    local prev = {
        Transparency = handle.Transparency,
        CanCollide = handle.CanCollide,
        Anchored = handle.Anchored,
    }

    pcall(function()
        handle.Transparency = 1
        handle.CanCollide = false
        handle.Anchored = false
    end)

    local clone = createDetachedClone(handle)
    if not clone then
        pcall(function()
            handle.Transparency = prev.Transparency
            handle.CanCollide = prev.CanCollide
            handle.Anchored = prev.Anchored
        end)
        return
    end

    trackedHandles[handle] = {
        clone = clone,
        orbiting = false,
        targetTorso = nil,
        angle = 0,
        spinAngle = 0,
        radius = MIN_ORBIT_RADIUS,
        yOffset = 0,
        prevProps = prev,
    }
end

local function stopTrackingHandle(handle)
    local data = trackedHandles[handle]
    if not data then return end
    pcall(function()
        if handle and handle.Parent then
            handle.Transparency = data.prevProps.Transparency or 0
            handle.CanCollide = data.prevProps.CanCollide or false
            handle.Anchored = data.prevProps.Anchored or false
        end
    end)
    pcall(function()
        if data.clone and data.clone.Parent then
            data.clone:Destroy()
        end
    end)
    trackedHandles[handle] = nil
end

local function findNearbyTorso(position, maxDist)
    local closest, bestDist = nil, maxDist
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character then
            local torso = getTorsoPart(p.Character)
            if torso then
                local success, d = pcall(function() return (torso.Position - position).Magnitude end)
                if success and d <= bestDist then
                    bestDist = d
                    closest = torso
                end
            end
        end
    end
    return closest
end
